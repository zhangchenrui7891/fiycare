cmake_minimum_required(VERSION 3.21)
project(FlightCareerAir VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(physics_check tests.cpp)
enable_testing()
add_test(NAME physics_and_career COMMAND physics_check)
if(WIN32)
 add_executable(FlightCareerAir WIN32 src/main.cpp)
 target_compile_definitions(FlightCareerAir PRIVATE UNICODE _UNICODE NOMINMAX)
 target_link_libraries(FlightCareerAir PRIVATE d3d12 d3d11 dxgi d3dcompiler user32 gdi32)
 if(MSVC)
  target_compile_options(FlightCareerAir PRIVATE /W4 /utf-8 /EHsc)
  target_compile_options(physics_check PRIVATE /W4 /utf-8)
 else()
  target_link_options(FlightCareerAir PRIVATE -municode)
 endif()
endif()
