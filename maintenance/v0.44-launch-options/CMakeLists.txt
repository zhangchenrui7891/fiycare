cmake_minimum_required(VERSION 3.22)
project(FlightCareerAirLaunchRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(launch_options_check launch_options_tests.cpp)
add_test(NAME exact_launch_options COMMAND launch_options_check)
if(WIN32)
 add_executable(windows_launch_check windows_launch_tests.cpp)
 target_link_libraries(windows_launch_check PRIVATE shell32)
 add_test(NAME actual_windows_command_line COMMAND windows_launch_check)
endif()
if(MSVC)
 add_compile_options(/utf-8)
 target_compile_options(launch_options_check PRIVATE /W4 /WX /utf-8 /EHsc)
 target_compile_options(windows_launch_check PRIVATE /W4 /WX /utf-8 /EHsc)
endif()
